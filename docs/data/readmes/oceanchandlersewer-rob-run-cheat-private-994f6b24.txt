⚙️ CONFIGURATION

Configuration file: config.json

Settings can be managed through the configuration file.

Example Configuration

{
  "aim": {
    "enabled": true,
    "preset": "ar",
    "fov": 120,
    "smoothing": 4.5,
    "bone": "head",
    "visible_only": true,
    "switch_delay_ms": 120
  },
  "overlay": {
    "players": true,
    "vehicles": true,
    "world": true,
    "max_distance": 300,
    "colors": {
      "enemy": "#FF4444",
      "ally": "#44FF44",
      "loot": "#FFD700"
    }
  },
  "radar": {
    "enabled": true,
    "size": 200,
    "zoom": 1.5,
    "rings": true,
    "offscreen_pins": true
  },
  "combat": {
    "mode": "aggressive",
    "recoil_control": true,
    "sway": 0.2
  },
  "ui": {
    "locale": "en",
    "dpi_scale": 1.0,
    "vsync": false,
    "fps_cap": 240
  }
}
